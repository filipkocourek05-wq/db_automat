<script>
  let dbData = [];
  let waitingForB = false;
  let waitingForPosition = false; // Nový stav pro čekání na č. pozice
  let tempA = "";
  let lastValidC = ""; 
  let seenCounts = {};
  let logs = [];

  // --- SÉRIOVÁ KOMUNIKACE S MAKROPADEM ---
  let serialPort;
  let serialWriter;

  async function connectSerial() {
    try {
      serialPort = await navigator.serial.requestPort();
      await serialPort.open({ baudRate: 9600 });
      const textEncoder = new TextEncoderStream();
      textEncoder.readable.pipeTo(serialPort.writable);
      serialWriter = textEncoder.writable.getWriter();
      
      const btn = document.getElementById('btnConnect');
      btn.textContent = "USB Připojeno";
      btn.style.background = "var(--ok)";
    } catch (e) {
      alert("Nepodařilo se připojit: " + e);
    }
  }

  async function sendToMacro(cmd) {
    if (serialWriter) {
      await serialWriter.write(cmd + "\n");
    } else {
      console.warn("Makropad není připojen! Simuluji: " + cmd);
    }
  }
  // ----------------------------------------

  const ioInput = document.getElementById('macroIO');
  const stepLabel = document.getElementById('stepLabel');
  const curAEl = document.getElementById('curA');
  const lastCEl = document.getElementById('lastCDisplay');

  function parseExcel() {
    const raw = document.getElementById('excelInput').value;
    if (!raw.trim()) { alert("Nejprve vlož data z Excelu!"); return; }

    const lines = raw.split(/\r?\n/);
    dbData = [];
    let totalRows = 0;

    for (let i = 0; i < lines.length; i++) {
      if (!lines[i].trim()) continue;
      totalRows++;
      const cols = lines[i].split('\t');
      const colA = (cols[0] || "").trim();
      const colB = (cols[1] || "").trim();
      const colF = (cols[5] || "").trim();
      const colL = (cols[11] || "").trim();
      if (colL.toLowerCase() === "aktuell") dbData.push({ a: colA, b: colB, f: colF, excelRow: i + 1 });
    }

    const fbInput = document.getElementById('fallbackVal');
    if (!fbInput.value && dbData.length > 0) fbInput.value = dbData[0].f;
    document.getElementById('excelStatus').innerHTML = `Načteno <b>${totalRows}</b> řádků. Aktivních: <b style="color:var(--ok)">${dbData.length}</b>.`;
    ioInput.focus();
  }

  ioInput.addEventListener('keydown', function(e) {
    if (e.key === 'Enter') {
      e.preventDefault();
      const val = ioInput.value.trim();
      ioInput.value = "";

      if (waitingForPosition) {
        // MAkropad právě vložil číslo chybové pozice z DB
        logs[0].dbPosText = "Chyba na pozici: " + val;
        waitingForPosition = false;
        renderLog();
        resetCycle();
        return;
      }

      if (!waitingForB) {
        tempA = val;
        waitingForB = true;
        curAEl.textContent = tempA || "(prázdné)";
        stepLabel.textContent = "ČEKÁM NA DATA B";
        stepLabel.style.color = "#f59e0b";
      } else {
        const tempB = val;
        waitingForB = false;
        const result = evaluateLookup(tempA, tempB);
        
        // Zde se HTML rozhoduje, co pošle makropadu do Arduina!
        if (result.status === "MISSING" || result.status === "MULTI") {
          sendToMacro("SKIP");
          waitingForPosition = true;
          stepLabel.textContent = "CHYBA -> ČEKÁM NA POZICI Z DB";
          stepLabel.style.color = "#ef4444";
        } else {
          sendToMacro("OK");
          stepLabel.textContent = "OK -> DATA C OZNAČENA PRO CTRL+C";
          stepLabel.style.color = "#22c55e";
          ioInput.value = result.outputC;
          ioInput.select();
        }
      }
    }
  });

  function evaluateLookup(valA, valB) {
    const seqInput = document.getElementById('nextSeqNum');
    const seq = parseInt(seqInput.value, 10) || 1;
    seqInput.value = seq + 1;

    const normA = valA.toLowerCase();
    const normB = valB.toLowerCase();
    const key = `${normA}|||${normB}`;

    seenCounts[key] = (seenCounts[key] || 0) + 1;
    const count = seenCounts[key];

    const matches = dbData.filter(item => item.a.toLowerCase() === normA && item.b.toLowerCase() === normB);
    const uniqueF = [...new Set(matches.map(m => m.f))];

    let status = "OK";
    let statusText = "OK";
    let outputC = "";
    const fallback = document.getElementById('fallbackVal').value;

    if (matches.length === 0) {
      status = "MISSING";
      outputC = lastValidC !== "" ? lastValidC : fallback;
      statusText = "CHYBÍ V EXCELU (Smazat)";
    } else if (matches.length > 1) {
      status = "MULTI";
      outputC = matches[0].f;
      lastValidC = outputC;
      lastCEl.textContent = lastValidC;
      statusText = `VÍCE SHOD (${matches.length}x)`;
    } else {
      outputC = matches[0].f;
      lastValidC = outputC;
      lastCEl.textContent = lastValidC;
      if (count > 1) { status = "DUP"; statusText = `DUPLICITA (${count}x)`; }
    }

    if (status !== "DUP" && count > 1) statusText += ` [Opak. ${count}x]`;

    logs.unshift({ seq, dbPosText: "Zjišťuji...", a: valA, b: valB, c: outputC, status, statusText });

    updateStats();
    renderLog();
    return { status: status, outputC: outputC };
  }

  function updateStats() {
    document.getElementById('stOk').textContent = logs.filter(l => l.status === 'OK').length;
    document.getElementById('stErr').textContent = logs.filter(l => l.status === 'MISSING').length;
    document.getElementById('stMulti').textContent = logs.filter(l => l.status === 'MULTI').length;
    document.getElementById('stDup').textContent = logs.filter(l => l.status === 'DUP').length;
  }

  function renderLog() {
    const onlyErr = document.getElementById('onlyErrors').checked;
    const tbody = document.getElementById('logBody');
    tbody.innerHTML = "";
    const filtered = onlyErr ? logs.filter(l => l.status !== 'OK') : logs;

    for (const item of filtered) {
      const tr = document.createElement('tr');
      tr.className = `row-${item.status}`;
      tr.innerHTML = `
        <td><b>#${item.seq}</b></td>
        <td>${escapeHtml(item.dbPosText)}</td>
        <td>${escapeHtml(item.a)}</td>
        <td>${escapeHtml(item.b)}</td>
        <td>${escapeHtml(item.c)}</td>
        <td><span class="pill pill-${item.status}">${item.statusText}</span></td>
      `;
      tbody.appendChild(tr);
    }
  }

  function resetCycle() {
    waitingForB = false;
    waitingForPosition = false;
    tempA = "";
    curAEl.textContent = "-";
    ioInput.value = "";
    stepLabel.textContent = "ČEKÁM NA DATA A";
    stepLabel.style.color = "var(--accent)";
    ioInput.focus();
  }

  function clearLog() {
    if (!confirm("Opravdu vymazat log?")) return;
    logs = []; seenCounts = {}; lastValidC = ""; lastCEl.textContent = "-";
    document.getElementById('nextSeqNum').value = 1;
    resetCycle(); updateStats(); renderLog();
  }

  function exportProblemsCSV() {
    const probs = logs.filter(l => l.status !== 'OK');
    if (probs.length === 0) { alert("Žádné problémové položky!"); return; }
    const text = "Poradi\tPozice_v_DB\tData_A\tData_B\tData_C\tProblem\n" + 
      probs.map(p => `#${p.seq}\t${p.dbPosText}\t${p.a}\t${p.b}\t${p.c}\t${p.statusText}`).join("\n");
    navigator.clipboard.writeText(text).then(() => alert("Kopírováno do schránky!"));
  }
  function escapeHtml(str) { return String(str).replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m])); }
</script>
